proc draw_square {x y size layer} {

  paint $layer
}

proc draw_con {x y w l layer} {
  ### ndif section.
  set x1    [expr $x-$l/2]
  set x2    [expr $x1+$l] 
  set y1    [expr $y-$w/2]
  set y2    [expr $y1+$w] 
  box $x1 $y1 $x2 $y2
  paint ndiff

  ### ndc section.
  set con_size  17
  set con_space 19
  set con_n [expr ($w+2) / ($con_size+$con_space)]

  set x1    [expr $x-$con_size/2]
  set x2    [expr $x1+$con_size]
  set y1    [expr $y-int((($con_n-1)/2.0)*($con_size+$con_space))-($con_size/2)]

  for {set i 0} {$i < $con_n} {incr i} {
    box $x1 [expr $y1+$i*($con_size+$con_space)] $x2 [expr $y1+$i*($con_size+$con_space)+$con_size]
    paint ndc
  }

  ### li section
  set li_over 8
  box [expr $x1-$li_over] [expr $y1-$li_over] [expr $x2+$li_over] [expr $y1+($i-1)*($con_size+$con_space)+$con_size+$li_over] 
  paint li

  ### m1 and mcon section
  if {![string compare $layer "m1"] || ![string compare $layer "m2"]} {
    set m1_over 8
    box [expr $x1-$m1_over] [expr $y1-$m1_over] [expr $x2+$m1_over] [expr $y1+($i-1)*($con_size+$con_space)+$con_size+$m1_over] 
    paint m1
    for {set i 0} {$i < $con_n} {incr i} {
      box $x1 [expr $y1+$i*($con_size+$con_space)] $x2 [expr $y1+$i*($con_size+$con_space)+$con_size]
      paint mcon
    }
  }

  ### m2 and m2con section
  if {![string compare $layer "m2"]} {
    set con_size_left   5
    set con_size_right 21
    set m2_over 8
    box [expr $x1-$m2_over] [expr $y1-$m2_over] [expr $x2+$m2_over] [expr $y1+($i-1)*($con_size+$con_space)+$con_size+$m2_over] 
    paint m2
    for {set i 0} {$i < $con_n} {incr i} {
      box [expr $x1-$con_size_left] [expr $y1+$i*($con_size+$con_space)-$con_size_left] [expr $x1+$con_size_right] [expr $y1+$i*($con_size+$con_space)+$con_size_right]
      paint m2con
    }
  }

}

proc draw_tran {x y w l nfing layer} {
  ### ndif section. Need con_size and con_space to match with con_test.
  set con_size  17
  set con_space 19

  set w_grid [expr int($w*100)]
  set l_grid [expr int($l*100)]
 
  set x1    [expr $x-(($con_size+$con_space-1)+$nfing*($con_size+$con_space-1+$l_grid))/2]
  set x2    [expr $x+(($con_size+$con_space-1)+$nfing*($con_size+$con_space-1+$l_grid))/2] 
  set y1    [expr $y-int($w_grid/2)]
  set y2    [expr $y1+$w_grid] 
  box $x1 $y1 $x2 $y2
  paint ndiff

  ### poly section.
  set poly_over  13

  for {set i 0} {$i < $nfing} {incr i} {
    set xpoly [expr $x1+$con_size+$con_space-1+$i*($con_size+$con_space-1+$l_grid)]
    box $xpoly [expr $y1-$poly_over] [expr $xpoly+$l_grid] [expr $y2+$poly_over]
    paint poly
  }

  ### con section
  for {set i 0} {$i < ($nfing+1)} {incr i} {
    set x_con [expr $x1+($con_size+$con_space-1)/2+$i*($con_size+$con_space-1+$l_grid)]
    draw_con $x_con $y $w_grid 35 $layer
  }
}

proc draw_tran_jog {x y wlist llist layerlist} {
  set poly_over  13
  set x_con      $x
  for {set i 0} {$i < [llength $layerlist]} {incr i} {
    ### con section
    if {$i == 0} {
      set wcon [lindex $wlist 0]
    } elseif {$i == [llength $layerlist]-1 } {
      set wcon [lindex $wlist [expr [llength $wlist]-1]]
    } else {
      set wcon [expr {min([lindex $wlist $i-1],[lindex $wlist $i])}]
    }
    draw_con $x_con $y [expr int(100*$wcon)] 35 [lindex $layerlist $i]

    ### poly section. And ndiff underneath.
    if {$i < [llength $llist]} {
      set x_con  [expr $x_con+18]
      set x_con2 [expr $x_con+100*[lindex $llist $i]]
      set wcon [lindex $wlist $i]
      box [expr $x_con-25] [expr $y-$wcon*50] [expr $x_con2+25] [expr $y+$wcon*50] 
      paint ndiff
      box $x_con [expr $y-$wcon*50-$poly_over] $x_con2 [expr $y+$wcon*50+$poly_over] 
      paint poly
      set x_con  [expr $x_con2+17]
    }
  }
}

proc draw_poly_con {x y con_n layer} {
  ### pc section.
  set con_size  17
  set con_space 19

  set x1    [expr $x-$con_size/2]
  set x2    [expr $x1+$con_size]
  set y1    [expr $y-int((($con_n-1)/2.0)*($con_size+$con_space))-($con_size/2)]

  for {set i 0} {$i < $con_n} {incr i} {
    box $x1 [expr $y1+$i*($con_size+$con_space)] $x2 [expr $y1+$i*($con_size+$con_space)+$con_size]
    paint pc
  }
  box [expr $x1-6] [expr $y1-8] [expr $x2+6] [expr $y1+($i-1)*($con_size+$con_space)+$con_size+8]
  paint poly
  paint li
  
  ### m1 and mcon section
  if {![string compare $layer "m1"] || ![string compare $layer "m2"]} {
    set m1_over 8
    box [expr $x1-$m1_over+2] [expr $y1-$m1_over] [expr $x2+$m1_over-2] [expr $y1+($i-1)*($con_size+$con_space)+$con_size+$m1_over] 
    paint m1
    for {set i 0} {$i < $con_n} {incr i} {
      box $x1 [expr $y1+$i*($con_size+$con_space)] $x2 [expr $y1+$i*($con_size+$con_space)+$con_size]
      paint mcon
    }
  }

  ### m2 and m2con section
  if {![string compare $layer "m2"]} {
    set con_size_left   5
    set con_size_right 21
    set m2_over 8
    box [expr $x1-$m2_over+2] [expr $y1-$m2_over] [expr $x2+$m2_over-2] [expr $y1+($i-1)*($con_size+$con_space)+$con_size+$m2_over] 
    paint m2
    for {set i 0} {$i < $con_n} {incr i} {
      box [expr $x1-$con_size_left] [expr $y1+$i*($con_size+$con_space)-$con_size_left] [expr $x1+$con_size_right] [expr $y1+$i*($con_size+$con_space)+$con_size_right]
      paint m2con
    }
  }

}

proc draw_tran_poly_con {x y w l nfing layer top bottom layer_pc} {
  ### draw_tran section. 
  draw_tran $x $y $w $l $nfing $layer

  ### draw_poly_con section. Need con_size and con_space to match with con_test.
  set con_size  17
  set con_space 19

  set w_grid [expr int($w*100)]
  set l_grid [expr int($l*100)]
 
  set x1    [expr $x-(($con_size+$con_space-1)+$nfing*($con_size+$con_space-1+$l_grid))/2]
  set y1    [expr $y-int($w_grid/2)-$bottom*($con_size+$con_space)/2-11]
  set y2    [expr $y+int($w_grid/2)+$top*($con_size+$con_space)/2+11]

  set poly_over  13

  if {$top > 0} {
    for {set i 0} {$i < $nfing} {incr i} {
      set xpoly [expr $x1+$con_size+$con_space-1+$i*($con_size+$con_space-1+$l_grid)]
      draw_poly_con [expr $xpoly+$l_grid/2] $y2 $top $layer_pc
    }
  }

  if {$bottom > 0} {
    for {set i 0} {$i < $nfing} {incr i} {
      set xpoly [expr $x1+$con_size+$con_space-1+$i*($con_size+$con_space-1+$l_grid)]
      draw_poly_con [expr $xpoly+$l_grid/2] $y1 $bottom $layer_pc
    }
  }

}

proc draw_via {x y wl space coverage nx ny layer1 layer2 layervia} {
  set wl_grid    [expr int($wl*100)]
  set space_grid [expr int($space*100)]
  set cover_grid [expr int($coverage*100)]

  set x1    [expr $x-int((($nx-1)/2.0)*($wl_grid+$space_grid))-($wl_grid/2)]
  set y1    [expr $y-int((($ny-1)/2.0)*($wl_grid+$space_grid))-($wl_grid/2)]

  for {set j 0} {$j < $ny} {incr j} {
    for {set i 0} {$i < $nx} {incr i} {
      box [expr $x1+$i*($wl_grid+$space_grid)] [expr $y1+$j*($wl_grid+$space_grid)] [expr $x1+$i*($wl_grid+$space_grid)+$wl_grid] [expr $y1+$j*($wl_grid+$space_grid)+$wl_grid] 
      paint $layervia
    }
  }

  box [expr $x1-$cover_grid] [expr $y1-$cover_grid] [expr $x1+$i*($wl_grid+$space_grid)-$space_grid+$cover_grid] [expr $y1+$j*($wl_grid+$space_grid)-$space_grid+$cover_grid]
  paint $layer1
  paint $layer2

}
